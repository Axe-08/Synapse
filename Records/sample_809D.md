### Description
Leha and Noora intend to visit $n$ towns ($1 \le n \le 3 \cdot 10^5$). 
Sightseeing in the $i$-th town is open only on days in the range $[l_i, r_i]$. 

They want to choose a visiting day $x_i \in [l_i, r_i]$ for each town and select a strictly increasing subsequence of towns $id_1 < id_2 < \dots < id_k$ such that the visiting dates are also strictly increasing: $x_{id_1} < x_{id_2} < \dots < x_{id_k}$.

Find the **maximum number of towns** they can visit ($k$).

// Time Complexity: O(N log N), Space Complexity: O(N)

STRUCT TreapNode:
  INT value, priority, left, right, lazy_tag

FUNCTION Update_Treap(l, r):
  // Split treap by values < l
  Split(root, l - 1, sub_left, sub_right)
  
  // Insert new candidate ending value 'l'
  Insert(sub_left, l)
  
  // Split remaining values by r - 1
  Split(sub_right, r - 1, sub_middle, sub_far_right)
  
  // Shift LIS values in [l, r-1] by +1
  Apply_Lazy_Add(sub_middle, +1)
  
  // Prune suboptimal tail element from sub_far_right
  Erase_Smallest_Element(sub_far_right)
  
  // Re-merge all treap components
  root = Merge(sub_left, Merge(sub_middle, sub_far_right))

#include <bits/stdc++.h>
using namespace std;

#define MAX_NODES (300000 + 10)

struct Treap_Node {
    int value, priority, left, right, lazy_add_tag;
};

Treap_Node Nodes[MAX_NODES];
int next_node_id, root_treap_id;
mt19937 random_generator(time(0) + clock());

int CreateNode(int val) {
    int id = next_node_id++;
    Nodes[id] = {val, (int)random_generator(), 0, 0, 0};
    return id;
}

void ApplyLazy(int node_id, int amount) {
    if (node_id != 0) {
        Nodes[node_id].value += amount;
        Nodes[node_id].lazy_add_tag += amount;
    }
}

void PushDownLazy(int node_id) {
    if (node_id != 0 && Nodes[node_id].lazy_add_tag != 0) {
        ApplyLazy(Nodes[node_id].left, Nodes[node_id].lazy_add_tag);
        ApplyLazy(Nodes[node_id].right, Nodes[node_id].lazy_add_tag);
        Nodes[node_id].lazy_add_tag = 0;
    }
}

void Split(int u_id, int w, int &x_id, int &y_id) {
    if (u_id == 0) { x_id = y_id = 0; return; }
    PushDownLazy(u_id);
    if (Nodes[u_id].value > w) {
        Split(Nodes[u_id].left, w, x_id, Nodes[u_id].left);
        y_id = u_id;
    } else {
        Split(Nodes[u_id].right, w, Nodes[u_id].right, y_id);
        x_id = u_id;
    }
}

int Merge(int u_id, int v_id) {
    if (u_id == 0) return v_id;
    if (v_id == 0) return u_id;
    PushDownLazy(u_id); PushDownLazy(v_id);
    if (Nodes[u_id].priority < Nodes[v_id].priority) {
        Nodes[u_id].right = Merge(Nodes[u_id].right, v_id);
        return u_id;
    } else {
        Nodes[v_id].left = Merge(u_id, Nodes[v_id].left);
        return v_id;
    }
}

int GetFront(int u_id) {
    if (Nodes[u_id].left == 0) return Nodes[u_id].value;
    PushDownLazy(u_id);
    return GetFront(Nodes[u_id].left);
}

void InsertIntoTreap(int &rt_id_ref, int x) {
    rt_id_ref = Merge(rt_id_ref, CreateNode(x));
}

void EraseFrontFromTreap(int &rt_id_ref) {
    if (rt_id_ref == 0) return;
    int val_to_remove = GetFront(rt_id_ref);
    int a_id, b_id;
    Split(rt_id_ref, val_to_remove, a_id, b_id);
    rt_id_ref = Merge(Nodes[a_id].left, Merge(Nodes[a_id].right, b_id));
}

void UpdateTreap(int l, int r) {
    int a_id, b_id, c_id, d_id;
    Split(root_treap_id, l - 1, a_id, b_id);
    InsertIntoTreap(a_id, l);
    Split(b_id, r - 1, c_id, d_id);
    ApplyLazy(c_id, 1);
    EraseFrontFromTreap(d_id);
    root_treap_id = Merge(a_id, Merge(c_id, d_id));
}

int GetTreapSize(int u_id) {
    if (u_id == 0) return 0;
    PushDownLazy(u_id);
    return GetTreapSize(Nodes[u_id].left) + GetTreapSize(Nodes[u_id].right) + 1;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N;
    cin >> N;
    next_node_id = 1;
    root_treap_id = CreateNode(0);

    for (int i = 0; i < N; i++) {
        int l, r;
        cin >> l >> r;
        UpdateTreap(l, r);
    }

    cout << GetTreapSize(root_treap_id) - 1 << "\n";
    return 0;
}
