### Description
Given an array $A$ of $N$ integers and $Q$ range queries $(L_i, R_i)$, for each query find the **minimum distance** between two identical elements within the subsegment $A[L_i \dots R_i]$. If no pair of identical elements exists, return $-1$.

$$d = \min \{ j - i \mid L_k \le i < j \le R_k \text{ and } A[i] = A[j] \}$$

  INT left, right, original_id

FUNCTION Main():

  READ Array A[1..N]

  // Map each value to its last seen index
  HASHMAP last_seen_index
  BUILD SegmentTree initialized with INF

  GROUP Queries BY right_endpoint ASCENDING

  FOR r = 1 TO N:
    val = A[r]
    IF val in last_seen_index:
      prev_idx = last_seen_index[val]
      dist = r - prev_idx
      // Update Segment Tree at position prev_idx with distance 'dist'
      SegmentTree_PointUpdate(prev_idx, dist)
    
    last_seen_index[val] = r

    // Answer all queries ending at 'r'
    FOR EACH query IN Queries_ending_at(r):

#include <bits/stdc++.h>
using namespace std;

const int INF = 1e9;

struct SegmentTree {
    int n;
    vector<int> tree;

    SegmentTree(int n) : n(n), tree(4 * n, INF) {}

    void update(int node, int start, int end, int idx, int val) {
        if (start == end) {
            tree[node] = min(tree[node], val);
            return;
        }
        int mid = (start + end) / 2;
        if (idx <= mid)
            update(2 * node, start, mid, idx, val);
        else
            update(2 * node + 1, mid + 1, end, idx, val);
        tree[node] = min(tree[2 * node], tree[2 * node + 1]);
    }

    int query(int node, int start, int end, int l, int r) {
        if (r < start || end < l) return INF;
        if (l <= start && end <= r) return tree[node];
        int mid = (start + end) / 2;
        return min(query(2 * node, start, mid, l, r),
                   query(2 * node + 1, mid + 1, end, l, r));
    }
};

struct Query {
    int l, r, id;
};

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, q;
    if (!(cin >> n >> q)) return 0;

    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<Query> queries(q);
    vector<vector<Query>> queries_at(n + 1);
    for (int i = 0; i < q; i++) {
        cin >> queries[i].l >> queries[i].r;
        queries[i].id = i;
        queries_at[queries[i].r].push_back(queries[i]);
    }

    SegmentTree st(n);
    map<int, int> last_pos;
    vector<int> ans(q);

    for (int r = 1; r <= n; r++) {
        if (last_pos.count(a[r])) {
            int prev = last_pos[a[r]];
            st.update(1, 1, n, prev, r - prev);
        }
        last_pos[a[r]] = r;

        for (auto &qry : queries_at[r]) {
            int res = st.query(1, 1, n, qry.l, r);
            ans[qry.id] = (res >= INF ? -1 : res);
        }
    }

    for (int i = 0; i < q; i++) {
        cout << ans[i] << "\n";
    }

    return 0;
}
